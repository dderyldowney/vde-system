vde_check_container_running () {
	local vm_name="$1" 
	local container_name
	container_name=$(_get_container_name "$vm_name") 
	local running_containers
	running_containers=$(docker ps --format "{{.Names}}" 2>/dev/null) 
	if [ $? -ne 0 ]
	then
		return $VDE_ERR_GENERAL
	fi
	if echo "$running_containers" | grep -q "^${container_name}$"
	then
		return $VDE_SUCCESS
	fi
	return $VDE_ERR_NOT_FOUND
}
vde_check_ssh_port () {
	local vm_name="$1" 
	local timeout="${2:-$VDE_SSH_TIMEOUT}" 
	local container_name
	container_name=$(_get_container_name "$vm_name") 
	local ssh_port
	ssh_port=$(docker port "$container_name" 22 2>/dev/null | grep -oE '0.0.0.0:[0-9]+' | cut -d: -f2) 
	if [ -z "$ssh_port" ]
	then
		return $VDE_ERR_GENERAL
	fi
	local elapsed=0 
	while [ $elapsed -lt $timeout ]
	do
		if (
				(
					zsh -c "exec 3<>/dev/tcp/localhost/$ssh_port" > /dev/null 2>&1
				)
			) 2> /dev/null
		then
			return $VDE_SUCCESS
		fi
		if command -v nc > /dev/null 2>&1
		then
			if nc -z localhost "$ssh_port" 2> /dev/null
			then
				return $VDE_SUCCESS
			fi
		fi
		local next_elapsed=$((elapsed + 2)) 
		if [ $next_elapsed -le $timeout ]
		then
			sleep 2
			elapsed=$next_elapsed 
		else
			elapsed=$timeout 
		fi
	done
	return $VDE_ERR_TIMEOUT
}

	done

vde_wait_for_container_healthy () {
	local vm_name="$1" 
	local max_wait="${2:-$VDE_DOCKER_TIMEOUT}" 
	local container_name
	container_name=$(_get_container_name "$vm_name") 
	local vm_type=$(get_vm_info type "$vm_name" 2>/dev/null) 
	local elapsed=0 
	local check_interval=1 
	local max_interval=5 
	echo "Waiting for $container_name to be healthy (max ${max_wait}s)..."
	while [ $elapsed -lt $max_wait ]
	do
		if vde_check_container_running "$vm_name"
		then
			if [ "$vm_type" = "lang" ]
			then
				if vde_check_ssh_login "$vm_name" 2
				then
					echo "✓ $container_name is healthy"
					return $VDE_SUCCESS
				fi
			else
				echo "✓ $container_name is healthy"
				return $VDE_SUCCESS
			fi
		fi
		sleep $check_interval
		elapsed=$((elapsed + check_interval)) 
		check_interval=$((check_interval * 2)) 
		[ $check_interval -gt $max_interval ] && check_interval=$max_interval 
		if [ $((elapsed % 30)) -eq 0 ] && [ $elapsed -gt 0 ]
		then
			echo "  [$elapsed/${max_wait}s] Still waiting for $container_name..."
		fi
	done
	echo "✗ $container_name did not become healthy within ${max_wait}s" >&2
	return $VDE_ERR_TIMEOUT
}
